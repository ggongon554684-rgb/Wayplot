// ===== WayPlot\AuthClient.cs
using System.Net;
using System.Net.Http.Json;
using WayPlot.domain.Contracts;

namespace WayPlot;

/// <summary>Talks to WayPlot.api. First sign-in needs a connection.</summary>
public sealed class AuthClient(HttpClient http)
{
    public async Task<LoginResult?> LoginAsync(string username, string password, CancellationToken ct = default)
    {
        var response = await http.PostAsJsonAsync("api/v1/auth/login", new LoginRequest(username, password), ct);
        if (response.StatusCode == HttpStatusCode.Unauthorized) return null;

        response.EnsureSuccessStatusCode();
        return await response.Content.ReadFromJsonAsync<LoginResult>(ct);
    }
}
// ===== WayPlot\LoginForm.cs
namespace WayPlot;

/// <summary>Temporary code-only login screen that proves Master routes a username to its tenant. Replaced by the real view later.</summary>
public sealed class LoginForm : Form
{
    private readonly AuthClient _auth;
    private readonly TextBox _user = new() { PlaceholderText = "Username", Width = 260 };
    private readonly TextBox _pass = new() { PlaceholderText = "Password", Width = 260, UseSystemPasswordChar = true };
    private readonly Button _go = new() { Text = "Sign in", Width = 260, Height = 36 };
    private readonly Label _msg = new() { AutoSize = false, Width = 260, Height = 70 };

    public LoginForm(AuthClient auth)
    {
        _auth = auth;
        Text = "WayPlot CRM";
        StartPosition = FormStartPosition.CenterScreen;
        FormBorderStyle = FormBorderStyle.FixedDialog;
        MaximizeBox = false;
        ClientSize = new Size(330, 300);

        var panel = new FlowLayoutPanel
        {
            Dock = DockStyle.Fill,
            FlowDirection = FlowDirection.TopDown,
            WrapContents = false,
            Padding = new Padding(30)
        };
        panel.Controls.AddRange(
        [
            new Label { Text = "Sign in", Font = new Font("Segoe UI", 16, FontStyle.Bold), AutoSize = true },
            _user, _pass, _go, _msg
        ]);
        Controls.Add(panel);

        AcceptButton = _go;
        _go.Click += async (_, _) => await SignInAsync();
    }

    private async Task SignInAsync()
    {
        _go.Enabled = false;
        try
        {
            var result = await _auth.LoginAsync(_user.Text.Trim(), _pass.Text);
            _msg.Text = result is null
                ? "Invalid username or password."
                : $"Signed in as {result.DisplayName} ({result.Role})\nWorkspace: {result.TenantName ?? "Platform"}, Tier {result.Tier}\nDatabase: {result.DatabaseName ?? "none"}";
        }
        catch (Exception ex)
        {
            _msg.Text = "Cannot reach the server: " + ex.Message;
        }
        finally
        {
            _go.Enabled = true;
        }
    }
}
// ===== WayPlot\Program.cs
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;

namespace WayPlot;

internal static class Program
{
    [STAThread]
    private static void Main()
    {
        ApplicationConfiguration.Initialize();

        var builder = Host.CreateApplicationBuilder(new HostApplicationBuilderSettings
        {
            ContentRootPath = AppContext.BaseDirectory
        });
        var apiUrl = builder.Configuration["Api:BaseUrl"] ?? "http://localhost:5039";

        builder.Services.AddSingleton(new HttpClient { BaseAddress = new Uri(apiUrl) });
        builder.Services.AddSingleton<AuthClient>();
        builder.Services.AddTransient<LoginForm>();

        using var host = builder.Build();
        Application.Run(host.Services.GetRequiredService<LoginForm>());
    }
}
